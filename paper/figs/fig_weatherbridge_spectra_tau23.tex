\newcommand{\spectralcurves}[3]{%
  \addplot[black, thin, dotted] coordinates {(80,1) (180,1)};
  \addplot[LinearGray, semithick, dash pattern=on 4pt off 2pt]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_linear.dat};
  \addplot[SwinBrown, semithick, densely dotted]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_fuxi.dat};
  \addplot[ModAFNOCyan, semithick, dash pattern=on 6pt off 2pt]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_modafno.dat};
  \addplot[SDYffPurple, semithick, dash pattern=on 3pt off 1pt on 1pt off 1pt]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_sdyff.dat};
  \addplot[PixelAttnGreen, thick, dash pattern=on 5pt off 1pt on 1pt off 1pt]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_pixelattn_vfi.dat};
  \addplot[DCAEOrange, thick, dash pattern=on 8pt off 2pt on 2pt off 2pt]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_weatherdcae.dat};
  \addplot[BridgeRed, very thick]
    table[x=ell,y=#3] {data/spectra_#1_tau#2_flow_spectral.dat};
}
